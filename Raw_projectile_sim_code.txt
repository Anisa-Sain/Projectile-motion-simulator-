        % Button pushed function: Scenario1Button
        function Scenario1ButtonPushed(app, event)
            cla(app.UIAxes);

            %making appropriate inputs visible to the user for scenario one
            app.HeightoflaunchpointmEditField.Visible = "on";        %making height label and edit field visible
            app.HeightoflaunchpointmEditFieldLabel.Visible = "on";

            app.DirectionofinitialvelocitydegreesEditField.Visible = "on";     %making direction of initial velocity label and edit field visible 
            app.DirectionofinitialvelocitydegreesEditFieldLabel.Visible ="on";

            app.MagnitudeofinitialvelocitymsEditField.Visible = "on";       %making magnitude of initial velocity label and edit field visible 
            app.MagnitudeofinitialvelocitymsEditFieldLabel.Visible = "on";

            app.InputdatahereLabel.Visible = "on";       %making "Input data here" label visible so the user knows ehat they should input

            app.PleasechoosescenarioonenowallortwowithwallLabel.Visible = "off";  %switching text from what scenario they want to choose, to explaining to click that scenario again
            app.PleaseclickchosenscenarioagaintolaunchobjectLabel.Visible = "on";

            app.HeightofhitmEditField.Visible = "off";   %making height of hit edit field and label invisible to the user
            app.HeightofhitmEditFieldLabel.Visible = "off";

            app.HorizontaldistancetravelledmEditField.Visible = "on";   %making horizontal distance travelled edit field and label visible 
            app.HorizontaldistancetravelledmEditFieldLabel.Visible ="on";

            app.WallheightmEditField.Visible = "off";    %making wall height label and edit field invisible to user
            app.WallheightmEditFieldLabel.Visible = "off";

            app.DistancefromlaunchpointtowallmEditField.Visible = "off";   %making distance from launch point to the wall edit field and label invisible
            app.DistancefromlaunchpointtowallmEditFieldLabel.Visible = "off";


            
            %defining variables Height, Direction and Magnitude by linking them to the corresponding edit fields
            height = app.HeightoflaunchpointmEditField.Value;
            direction = app.DirectionofinitialvelocitydegreesEditField.Value; 
            magnitude = app.MagnitudeofinitialvelocitymsEditField.Value;

 
            
            %calculating Ti
            Ti = 0;     %defining Ti at the start
            %assigning variables(0.5gTi^2 + U_verticalTi + H = 0) in the equation to a b and c 
            a = -4.905;    %half of -9.81
            b = magnitude*sind(direction);
            c = height;

            %calculation is done in steps to avoid error 
            discriminant = (b)^2-(4*a*c);  %calculating the discriminant 
            dsquare = sqrt(discriminant);  %calculating the square root of the discriminant 
            quadPlus = (-b+(dsquare))/(2*a);  %using the quadratic formula to calculate the positive and negative results of the equation
            quadNeg = (-b-(dsquare))/(2*a);
            if quadNeg < 0           %checking and assigning the positive results to Ti 
                Ti = quadPlus;
            else
                Ti = quadNeg;
            end
            %calulating vertical and horizontal initial velocity
            U_vertical = magnitude*sind(direction); 
            U_horizontal = magnitude*cosd(direction);


            %defining horizontal and vertical components of graph
            S_horizontal = U_horizontal*Ti;
            S_vertical = (U_horizontal*Ti) +(0.5 *(-9.81)*(Ti.^2))+ height;

            %assinging variables to Edit Fields
            app.HorizontaldistancetravelledmEditField.Value = S_horizontal;
            app.TimeofflightsEditField.Value = Ti;
            app.MaximumHeightmEditField.Value = ((U_vertical.^2)/19.62)+height;

            
              
            %creating animated lines of trajectory 
            t = linspace(0,Ti,100);      %generates a linear spaced vector for time 

            Anline = animatedline(app.UIAxes); %defining the animated line and assigning it to AnLine

            x =(U_horizontal *t);    %defining x values and y values, using time to link them
            y = (U_vertical*t)+((-4.905)*(t.^2))+ height;

            for i = 1:length(t)                %creating a loop where it loops from 1 to the length of t
                if isvalid(Anline)
                    addpoints(Anline,x(i),y(i));   %adding coordinates to create the animated line
                else
                    break
                end
                drawnow;                       %drawing each point
            end

           hold(app.UIAxes,"on")        %making sure the previous line made stays on the graph  
           quiver(app.UIAxes, 0, height, U_horizontal, 0, "AutoScale", "off") %showing the initial horizontal and vertical velocity in the form of vectors 
           quiver(app.UIAxes, 0, height, 0, U_vertical, "AutoScale", "off")


        end

        function Scenario2ButtonPushed(app, event)

            %making appropriate inputs visible to the user 
            app.HeightoflaunchpointmEditField.Visible = "on";   %making height label and edit field visible
            app.HeightoflaunchpointmEditFieldLabel.Visible = "on";

            app.DirectionofinitialvelocitydegreesEditFieldLabel.Visible ="on";

            app.MagnitudeofinitialvelocitymsEditField.Visible = "on";     %making magnitude of initial velocity label and edit field visible 
            app.MagnitudeofinitialvelocitymsEditFieldLabel.Visible = "on";

            app.InputdatahereLabel.Visible = "on";     %making "Input data here" label visible so the user knows ehat they should input

            app.DistancefromlaunchpointtowallmEditField.Visible = "on";  %making distance from the launch point to the wall label and edit field visible 
            app.DistancefromlaunchpointtowallmEditFieldLabel.Visible = "on";

            app.WallheightmEditField.Visible = "on";    %making the wall height label and edit field visible 
            app.WallheightmEditFieldLabel.Visible = "on";

            %switching text on the screen from "Please choose scenario one(no wall) or scenario two(with wall):" to
            % "Please click chosen scenario again to launch object"
            app.PleasechoosescenarioonenowallortwowithwallLabel.Visible = "off";
            app.PleaseclickchosenscenarioagaintolaunchobjectLabel.Visible = "on";
            

            
            
            %defining variables Height, Direction, Magnitude, wall_height, and wall_distance by linking them to the corresponding edit fields
            height = app.HeightoflaunchpointmEditField.Value;
            direction = app.DirectionofinitialvelocitydegreesEditField.Value; 
            magnitude = app.MagnitudeofinitialvelocitymsEditField.Value;
            wall_height = app.WallheightmEditField.Value;
            wall_distance = app.DistancefromlaunchpointtowallmEditField.Value;


            %plotting wall, changing the line width to 2 and the colour to red
            wall = plot(app.UIAxes, [wall_distance, wall_distance],[0, wall_height],"Color","r", "LineWidth",2);


            %calculating Ti
            Ti = 0;
            %assigning variables(0.5gTi^2 + U_verticalTi + H = 0) in the equation to a b and c 
            a = -4.905;
            b = magnitude*sind(direction);
            c = height;
            %calculation is done in steps to avoid error 
            discriminant = (b)^2-(4*a*c);  %calculating the discriminant 
            dsquare = sqrt(discriminant);  %calculating the square root of the discriminant 
            quadPlus = (-b+(dsquare))/(2*a);  %using the quadratic formula to calculate the results of the equation
            quadNeg = (-b-(dsquare))/(2*a);
            if quadNeg < 0           %checking and assigning the positive results to Ti 
                Ti = quadPlus;
            else
                Ti = quadNeg;
            end


            %calulating vertical and horizontal initial velocity
            U_vertical = magnitude*sind(direction); 
            U_horizontal = magnitude*cosd(direction);


            %defining horizontal and vertical components of graph
            S_horizontal = U_horizontal*Ti;
            S_vertical = (U_horizontal*Ti) +(0.5 *(-9.81)*(Ti.^2))+ height;

            %calculating Tw
            Tw = wall_distance/ U_horizontal;

            %calculting the altitude at the distance from the launch point to the wall 
            Syw = (U_vertical*Tw)+(-4.905*(Tw.^2))+height;


    

            %determining whether to use Tw or Ti by seeing if the altitudeis above zero and less than the
            %wall height by using logical AND
           
            if Syw > 0 && Syw < wall_height
                Ti = Tw;                     %Tw holds the value of the time that the object hits the wall
                app.HeightofhitmEditField.Visible = "on";   %showing the user the height of hit on the wall 
                app.HeightofhitmEditFieldLabel.Visible = "on";
                Height_of_hit = (U_vertical*Ti)+(-4.905*(Ti.^2))+height;
                app.HeightofhitmEditField.Value = Height_of_hit;
                app.HorizontaldistancetravelledmEditField.Visible = "off"; %hiding the horizontal distance from the user 
                app.HorizontaldistancetravelledmEditFieldLabel.Visible = "off";
                

            end

            
            %assinging variables to Edit Fields
            app.HorizontaldistancetravelledmEditField.Value = S_horizontal;
            app.TimeofflightsEditField.Value = Ti;
            app.MaximumHeightmEditField.Value = ((U_vertical.^2)/19.62)+height;
            

            %creating animated lines of trajectory 
            t = linspace(0,Ti,100);      %generates a linear spaced vector for time 

            Anline = animatedline(app.UIAxes); %defining the animated line and assigning it to AnLine

            x =(U_horizontal *t);    %defining x values and y values, using time to link them
            y = (U_vertical*t)+((-4.905)*(t.^2))+ height;

            for i = 1:length(t)                %creating a loop where it loops from 1 to the length of t
                if isvalid(Anline)
                    addpoints(Anline,x(i),y(i));   %adding coordinates to create the animated line
                else
                    break
                end
                drawnow;                       %drawing each point
            end
           hold(app.UIAxes,"on")         %%making sure the previous line made stays on the graph 
           quiver(app.UIAxes, 0, height, U_horizontal, 0, "AutoScale", "off") %showing the initial horizontal and vertical velocity in the form of vectors 
           quiver(app.UIAxes, 0, height, 0, U_vertical, "AutoScale", "off")


        end

        % Value changing function: PointonlineSlider
        function PointonlineSliderValueChanging(app, event)
            changingValue = event.Value;

            %defining the horizontal distance and height of hit on the wa;;
            horo_dis = app.HorizontaldistancetravelledmEditField.Value;
            hit_wall = app.HeightofhitmEditField.Value;
            

            %defining the limits of the slider as 0 to the horizontal
            %distance
            app.PointonlineSlider.Limits = [0 horo_dis];
            
            %defining variables to be calculated from the point on slider
            x_coords = app.XcoordinatesEditField.Value;
            y_coords = app.YcoordinatesEditField.Value;
            time_passed = app.TimepassedEditField.Value;
            current_v = app.CurrentvelocityEditField.Value;
            direction_slider = app.DirectiondegreesEditField.Value;

            %defining variables previously used in other functions
            magnitude = app.MagnitudeofinitialvelocitymsEditField.Value;
            direction = app.DirectionofinitialvelocitydegreesEditField.Value; 
            U_vertical = magnitude*sind(direction);
            U_horizontal = magnitude *cosd(direction);
            height = app.HeightoflaunchpointmEditField.Value;
            wall_height = app.WallheightmEditField.Value;
            wall_distance = app.DistancefromlaunchpointtowallmEditField.Value;
            
            
            %calculating Ti
            Ti = 0;
            %assigning variables(0.5gTi^2 + U_verticalTi + H = 0) in the equation to a b and c 
            a = -4.905;
            b = magnitude*sind(direction);
            c = height;
            %calculation is done in steps to avoid error 
            discriminant = (b)^2-(4*a*c);  %calculating the discriminant 
            dsquare = sqrt(discriminant);  %calculating the square root of the discriminant 
            quadPlus = (-b+(dsquare))/(2*a);  %using the quadratic formula to calculate the results of the equation
            quadNeg = (-b-(dsquare))/(2*a);
            if quadNeg < 0           %checking and assigning the positive results to Ti 
                Ti = quadPlus;
            else
                Ti = quadNeg;
            end

            %calculating Tw
            Tw = wall_distance/ U_horizontal;

            %calculting the altitude at the distance from the launch point to the wall 
            Syw = (U_vertical*Tw)+(-4.905*(Tw.^2))+height;


            %determining whether to use Tw or Ti by seeing if the altitudeis above zero and less than the wall height by using logical AND
            if Syw > 0 && Syw < wall_height
                Ti = Tw;  
            end


            
            %taking value from the slider and assigning to the x coordinates 
            x_coords = app.PointonlineSlider.Value;

            %calculating y coordinates 

            y_coords = ((U_vertical*(x_coords/U_horizontal))+((-4.905)*((x_coords/U_horizontal).^2))+ height) ;
            
            %assigning calculated x and y coordinates to their corresponding edit fields
            app.XcoordinatesEditField.Value = x_coords;
            app.YcoordinatesEditField.Value = y_coords;
            
            %calculating time passed 
            time_passed = x_coords/ U_horizontal;
            %asssigning time_passed to corressponding edit field
            app.TimepassedEditField.Value = time_passed;

            %calculating current velocity using the equation V = U + at
            current_v = magnitude + (-9.81*time_passed);
            current_v  = magnitude +time_passed;
            %assigning current_v to its edit field 
            app.CurrentvelocityEditField.Value = current_v;


               
            %calculating horizontal and vertical displacement 
            S_horizontal = U_horizontal*(x_coords/U_horizontal);
            S_vertical = (U_horizontal*(x_coords/U_horizontal)) +(0.5 *(-9.81)*((x_coords/U_horizontal).^2))+ height;
            %calculating direction
            direction_slider = atand(S_vertical/S_horizontal)-45;
            

            %stopping the slider from going past the horizontal distance travelled for scenario two
            if time_passed>=Ti
                horo_dis = U_horizontal *Ti;  %calulating new horizontal distance
                app.HorizontaldistancetravelledmEditField.Value = horo_dis;  %updating the horizontal distance edit field 
            end


            %updating all edit fields corressponding to the slider with updated values
            app.DirectiondegreesEditField.Value = direction_slider;
            app.XcoordinatesEditField.Value = x_coords;
            app.YcoordinatesEditField.Value = y_coords;
            app.TimepassedEditField.Value = time_passed;
            app.CurrentvelocityEditField.Value = current_v;


        end
    end
